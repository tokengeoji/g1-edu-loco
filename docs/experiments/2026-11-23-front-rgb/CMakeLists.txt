cmake_minimum_required(VERSION 3.10)
project(front_color)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core videoio)
add_executable(front_color_bridge bridge.cpp)
target_compile_features(front_color_bridge PRIVATE cxx_std_17)
ament_target_dependencies(front_color_bridge rclcpp sensor_msgs)
target_link_libraries(front_color_bridge ${OpenCV_LIBS})
